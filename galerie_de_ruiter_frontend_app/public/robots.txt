# Everything behind an account is closed to crawlers: those routes answer with the sign-in
# page, so crawling them only spends the crawl budget. What a visitor reaches without an
# account - the landing page, the gallery story, the pop-up's address, the concierge chat, the
# collection list and a piece's own page - is listed in sitemap.xml, whose URL is appended below
# at build time by scripts/generate-sitemap.mjs.
#
# The collection is open: the list and a piece's own page are public, and the API's
# GET /api/antiques gives the sitemap script the ids to list. The actions on a piece - saving it,
# putting it in the basket, liking it - are what still ask for an account.
User-agent: *

Disallow: /admin
Disallow: /cart
Disallow: /wishlist
Disallow: /profile
Disallow: /preferences
Disallow: /logout

Allow: /
Allow: /antiques
Allow: /about
Allow: /map
Allow: /chat

# The sign-in and registration pages stay crawlable on purpose: they carry a noindex tag, and
# a crawler that is not allowed to fetch them would never see it.
#
# The `Sitemap:` line is appended below at build time by scripts/generate-sitemap.mjs, from the site
# origin in site.config.mjs, so this file stays origin-independent. It takes an absolute URL, which is
# why it is not written here by hand.
#
# An `Agentmap:` line is not added, and one left by an earlier build is removed. The Agent Discovery
# manifest is still served at /.well-known/ard.json; the document advertises it with <link rel="ard">,
# the relation a conforming consumer must honour, so robots.txt does not have to name it - and a line
# kept only because a build once wrote it would outlive whoever decided against it.
